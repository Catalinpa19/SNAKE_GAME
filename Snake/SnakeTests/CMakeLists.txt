cmake_minimum_required(VERSION 3.24)
project(SnakeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg Test)
set(GMOCK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../packages/gmock.1.11.0/lib/native" CACHE PATH "GoogleMock NuGet native directory")
if(NOT EXISTS "${GMOCK_ROOT}/src/gtest/gtest-all.cc")
    message(FATAL_ERROR "Restore SnakeTests/packages.config using NuGet, or set GMOCK_ROOT.")
endif()
add_library(snake_gmock STATIC "${GMOCK_ROOT}/src/gtest/gtest-all.cc" "${GMOCK_ROOT}/src/gmock/gmock-all.cc")
set_target_properties(snake_gmock PROPERTIES AUTOMOC OFF)
target_include_directories(snake_gmock PUBLIC "${GMOCK_ROOT}/include" PRIVATE "${GMOCK_ROOT}/src/gtest" "${GMOCK_ROOT}/src/gmock")
set(UI "${CMAKE_CURRENT_SOURCE_DIR}/../SnakeUI")
file(GLOB UI_SOURCES CONFIGURE_DEPENDS "${UI}/app/*.cpp" "${UI}/app/*.h" "${UI}/widgets/*.cpp" "${UI}/widgets/*.h" "${UI}/screens/*.cpp" "${UI}/screens/*.h")
list(REMOVE_ITEM UI_SOURCES "${UI}/app/SoundPlayer.cpp")
add_executable(SnakeTests main.cpp test.cpp ProfileTests.cpp TestSoundPlayer.cpp ${UI_SOURCES} "${UI}/SnakeUI.qrc")
target_include_directories(SnakeTests PRIVATE "${UI}")
target_link_libraries(SnakeTests PRIVATE snake_gmock Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg Qt6::Test)
if(MSVC)
    target_compile_options(SnakeTests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME SnakeUiTests COMMAND SnakeTests)
set_tests_properties(SnakeUiTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(WIN32)
    get_target_property(QT_QMAKE Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(QT_BIN "${QT_QMAKE}" DIRECTORY)
    set_tests_properties(SnakeUiTests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${QT_BIN}")
endif()
